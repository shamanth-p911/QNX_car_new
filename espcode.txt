#include <Arduino.h>

// ======================================
// PIN DEFINITIONS
// ======================================

#define LEFT_BUTTON   18
#define RIGHT_BUTTON  19

#define LEFT_LED      25
#define RIGHT_LED     26

#define POT_PIN       34

#define TRIG_PIN      32
#define ECHO_PIN      33

// ======================================
// VARIABLES
// ======================================

char command[64];
int commandIndex = 0;

unsigned long lastSend = 0;

const unsigned long SEND_INTERVAL = 50;

// ======================================
// ULTRASONIC SENSOR
// ======================================

int readDistance()
{
    // Clear trigger
    digitalWrite(TRIG_PIN, LOW);
    delayMicroseconds(2);

    // Send 10 microsecond trigger
    digitalWrite(TRIG_PIN, HIGH);
    delayMicroseconds(10);
    digitalWrite(TRIG_PIN, LOW);

    // Measure echo duration
    unsigned long duration =
        pulseIn(ECHO_PIN, HIGH, 30000UL);

    // No echo received
    if (duration == 0)
    {
        return -1;
    }

    // Calculate distance in cm
    int distance = (int)(
        duration * 0.0343 / 2.0
    );

    return distance;
}

// ======================================
// SERIAL COMMAND PROCESSING
// ======================================

void processCommand(const char *cmd)
{
    int left, right;
    char extra;

    // LED command from QNX
    // Example: LED:1,0

    if (sscanf(
            cmd,
            "LED:%d,%d%c",
            &left,
            &right,
            &extra
        ) == 2)
    {
        if ((left == 0 || left == 1) &&
            (right == 0 || right == 1))
        {
            digitalWrite(
                LEFT_LED,
                left ? HIGH : LOW
            );

            digitalWrite(
                RIGHT_LED,
                right ? HIGH : LOW
            );
        }

        return;
    }

    // Ultrasonic polling command from QNX
    // Example: US?

     // Ultrasonic polling command from QNX: "U?"
    // Reply format QNX parses: "D:<cm>"
    if (strcmp(cmd, "U?") == 0)
    {
        int distance = readDistance();
        Serial.printf("D:%d\n", distance);
        return;
    }
}

// ======================================
// SETUP
// ======================================

void setup()
{
    Serial.begin(115200);

    // Pushbuttons
    pinMode(
        LEFT_BUTTON,
        INPUT_PULLUP
    );

    pinMode(
        RIGHT_BUTTON,
        INPUT_PULLUP
    );

    // LEDs
    pinMode(
        LEFT_LED,
        OUTPUT
    );

    pinMode(
        RIGHT_LED,
        OUTPUT
    );

    digitalWrite(
        LEFT_LED,
        LOW
    );

    digitalWrite(
        RIGHT_LED,
        LOW
    );

    // Potentiometer
    pinMode(POT_PIN, INPUT);

    analogReadResolution(12);

    // Ultrasonic sensor
    pinMode(
        TRIG_PIN,
        OUTPUT
    );

    pinMode(
        ECHO_PIN,
        INPUT
    );

    digitalWrite(
        TRIG_PIN,
        LOW
    );
}

// ======================================
// MAIN LOOP
// ======================================

void loop()
{
    // Send button and potentiometer
    // readings to QNX every 50 ms

    if (millis() - lastSend >= SEND_INTERVAL)
    {
        lastSend = millis();

        int left =
            digitalRead(LEFT_BUTTON);

        int right =
            digitalRead(RIGHT_BUTTON);

        int pot =
            analogRead(POT_PIN);

        Serial.printf(
            "L:%d,R:%d,P:%d\n",
            left,
            right,
            pot
        );
    }

    // Receive commands from QNX

    while (Serial.available())
    {
        char ch = Serial.read();

        if (ch == '\r')
        {
            continue;
        }

        if (ch == '\n')
        {
            command[commandIndex] = '\0';

            processCommand(command);

            commandIndex = 0;
        }
        else if (
            commandIndex <
            (int)sizeof(command) - 1
        )
        {
            command[commandIndex++] = ch;
        }
        else
        {
            commandIndex = 0;
        }
    }
}
